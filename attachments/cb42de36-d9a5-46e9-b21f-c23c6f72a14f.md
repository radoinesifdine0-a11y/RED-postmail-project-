**Prérequis:** Avoir complété les Parts 1 et 2

**Objectif:** Créer un serveur MCP (Model Context Protocol) permettant à Claude et autres LLMs d'interagir avec l'API Store Parcels via des outils structurés.

---

## 1. Introduction au MCP

### 1.1 Qu'est-ce que MCP ?

**Model Context Protocol (MCP)** est un standard ouvert développé par Anthropic permettant aux LLMs d'interagir avec des systèmes externes de manière structurée et sécurisée.

**Concepts clés:**
- **Tools**: Fonctions que le LLM peut appeler (comme `search_parcels`, `create_parcel`)
- **Resources**: Données que le LLM peut lire (comme une liste de stores)
- **Prompts**: Templates de prompts prédéfinis (optionnel)

**Architecture:**
```
┌─────────────┐     ┌─────────────┐     ┌─────────────┐
│   Claude    │────►│ MCP Server  │────►│ Store API   │
│   (Client)  │◄────│ (TypeScript)│◄────│ (Java/Spring)│
└─────────────┘     └─────────────┘     └─────────────┘
```

### 1.2 Pourquoi un Serveur MCP ?

- **Interaction naturelle**: Permet à Claude de manipuler les parcels via conversation
- **Abstraction**: Cache la complexité de l'API REST
- **Sécurité**: Contrôle précis des opérations autorisées
- **Observabilité**: Logging centralisé des interactions LLM

```typescript
import { Tool } from '@modelcontextprotocol/sdk/types.js';

export const searchParcelsTool: Tool = {
  name: 'search_parcels',
  description: `Search for parcels in the store parcel system.

Use this tool to:
- Find parcels at a specific store (deliveryStoreId)
- Look up a customer's parcels by name
- Filter by parcel status
- Get details for a specific parcel or order

At least one search criteria should be provided for best results.
Results are paginated (max 50 per page).`,
  inputSchema: {
    type: 'object',
    properties: {
      store_id: {
        type: 'string',
        pattern: '^ST\\d{3}$',
        description: 'Store ID (format: ST001-ST999) - maps to deliveryStoreId'
      },
      customer_first_name: {
        type: 'string',
        description: "Customer's first name (uses LIKE operator for partial match)"
      },
      customer_last_name: {
        type: 'string',
        description: "Customer's last name (uses LIKE operator for partial match)"
      },
      status: {
        type: 'string',
        enum: ['PENDING', 'IN_TRANSIT', 'READY_FOR_PICKUP', 'PICKED_UP', 'CANCELLED', 'RETURNED', 'EXPIRED', 'DELIVERED'],
        description: 'Parcel status filter (parcelStatus)'
      },
      parcel_id: {
        type: 'string',
        description: 'Exact parcel ID'
      },
      order_id: {
        type: 'string',
        description: 'Customer order ID (customerOrderId)'
      },
      page: {
        type: 'integer',
        minimum: 1,
        default: 1,
        description: 'Page number (1-indexed)'
      },
      page_size: {
        type: 'integer',
        minimum: 10,
        maximum: 50,
        default: 20,
        description: 'Results per page (10-50)'
      }
    }
  }
};

export async function handleSearchParcels(
  args: z.infer<typeof SearchParcelsInputSchema>,
  apiClient: StoreApiClient
): Promise<string> {
  // Build criteria array for the API
  const criterias: Criteria[] = [];

  if (args.store_id) {
    criterias.push({ field: 'deliveryStoreId', operator: 'EQ', value: args.store_id });
  }
  if (args.customer_first_name) {
    criterias.push({ field: 'customer.firstName', operator: 'LIKE', value: args.customer_first_name });
  }
  if (args.customer_last_name) {
    criterias.push({ field: 'customer.lastName', operator: 'LIKE', value: args.customer_last_name });
  }
  if (args.status) {
    criterias.push({ field: 'parcelStatus', operator: 'EQ', value: args.status });
  }
  if (args.parcel_id) {
    criterias.push({ field: 'parcelId', operator: 'EQ', value: args.parcel_id });
  }
  if (args.order_id) {
    criterias.push({ field: 'customerOrderId', operator: 'EQ', value: args.order_id });
  }

  const response = await apiClient.searchParcels({
    criterias,
    pagination: { page: args.page ?? 1, pageSize: args.page_size ?? 20 },
    sorts: [{ index: '0', field: 'expirationDateTime', order: 'asc' }]
  });

  return formatSearchResults(response);
}

function formatSearchResults(response: SearchResult): string {
  const { parcels, totalElements, currentPage, totalPages, pageSize } = response;

  if (!parcels || parcels.length === 0) {
    return 'No parcels found matching the search criteria.';
  }

  let result = `Found ${totalElements} parcels (showing ${parcels.length} on page ${currentPage} of ${totalPages}):\n\n`;

  for (const parcel of parcels) {
    result += `**${parcel.parcelId}** - ${parcel.parcelStatus}`;
    if (parcel.isExpired) result += ' (EXPIRED)';
    result += '\n';
    result += `  Customer: ${parcel.customer.firstName} ${parcel.customer.lastName}\n`;
    result += `  Store: ${parcel.deliveryStoreId}\n`;
    result += `  Order: ${parcel.customerOrderId}\n`;
    result += `  Expires: ${new Date(parcel.expirationDate).toLocaleDateString()}\n`;
    result += `  Paid: ${parcel.isPaid ? 'Yes' : 'No'}\n`;
    if (parcel.customer.email) result += `  Email: ${parcel.customer.email}\n`;
    result += '\n';
  }

  return result;
}
```

```typescript
export async function handleCreateParcel(
  args: CreateParcelArgs,
  apiClient: StoreApiClient
): Promise<string> {
  // Default expiration: 14 days from now
  const expirationDate = args.expiration_date
    ? new Date(args.expiration_date)
    : new Date(Date.now() + 14 * 24 * 60 * 60 * 1000);

  const request: CreateParcelRequest = {
    parcelId: args.parcel_id,
    customerOrderId: args.order_id,
    deliveryStoreId: args.store_id,
    parcelStatus: args.status || 'READY_FOR_PICKUP',
    expirationDate: expirationDate.toISOString(),
    isPaid: args.is_paid ?? false,
    customer: {
      firstName: args.customer_first_name,
      lastName: args.customer_last_name,
      email: args.customer_email,
      phoneNumber: args.customer_phone
    },
    storageLocation: {
      storeId: args.store_id,
      locationCode: args.storage_location_code
    }
  };

  try {
    const parcel = await apiClient.createParcel(request);

    return `
✅ **Parcel Created Successfully**

**Parcel ID:** ${parcel.parcelId}
**Order ID:** ${parcel.customerOrderId}
**Store:** ${parcel.deliveryStoreId}
**Customer:** ${parcel.customer.firstName} ${parcel.customer.lastName}
**Status:** ${parcel.parcelStatus}
**Expires:** ${new Date(parcel.expirationDate).toLocaleDateString()}
**Paid:** ${parcel.isPaid ? 'Yes' : 'No'}

The parcel is now registered in the system.
`.trim();
  } catch (error: any) {
    if (error.status === 409) {
      return `❌ **Error:** Parcel with ID ${args.parcel_id} already exists.`;
    }
    throw error;
  }
}
```

### 4.4 Tool: update_parcel_status

**Description**: Met à jour le statut d'un parcel

```typescript
export const updateParcelStatusTool: Tool = {
  name: 'update_parcel_status',
  description: `Update the status of a parcel.

Valid status transitions (from Part 2 state machine):
- PENDING → IN_TRANSIT, CANCELLED
- IN_TRANSIT → READY_FOR_PICKUP, CANCELLED
- READY_FOR_PICKUP → PICKED_UP, EXPIRED, CANCELLED
- PICKED_UP → DELIVERED, CANCELLED
- EXPIRED → RETURNED, CANCELLED

Terminal statuses (no further changes): DELIVERED, RETURNED, CANCELLED

Uses the endpoint PUT /store_parcels/parcel/{parcelId}`,
  inputSchema: {
    type: 'object',
    properties: {
      parcel_id: {
        type: 'string',
        description: 'The parcel ID to update'
      },
      new_status: {
        type: 'string',
        enum: ['IN_TRANSIT', 'READY_FOR_PICKUP', 'PICKED_UP', 'DELIVERED', 'EXPIRED', 'RETURNED', 'CANCELLED'],
        description: 'The new status'
      },
      mark_as_paid: {
        type: 'boolean',
        description: 'Mark the parcel as paid (optional)'
      }
    },
    required: ['parcel_id', 'new_status']
  }
};

export async function handleUpdateParcelStatus(
  args: UpdateParcelArgs,
  apiClient: StoreApiClient
): Promise<string> {
  try {
    const updateRequest: UpdateParcelRequest = {
      parcelStatus: args.new_status
    };

    if (args.mark_as_paid !== undefined) {
      updateRequest.isPaid = args.mark_as_paid;
    }

    const parcel = await apiClient.updateParcel(args.parcel_id, updateRequest);

    return `
✅ **Parcel Updated Successfully**

**Parcel ID:** ${parcel.parcelId}
**New Status:** ${parcel.parcelStatus}
${args.mark_as_paid !== undefined ? `**Payment Status:** ${parcel.isPaid ? 'Paid' : 'Not paid'}` : ''}
`.trim();

  } catch (error: any) {
    if (error.status === 400 && error.body?.title === 'Invalid Status Transition') {
      const { currentStatus, targetStatus, allowedTransitions } = error.body;
      return `
❌ **Invalid Status Transition**

Cannot change status from **${currentStatus}** to **${targetStatus}**.

Allowed transitions from ${currentStatus}:
${allowedTransitions?.length > 0
  ? allowedTransitions.map((s: string) => `- ${s}`).join('\n')
  : '- None (terminal status)'}
`.trim();
    }

    if (error.status === 404) {
      return `❌ **Error:** Parcel ${args.parcel_id} not found.`;
    }

    throw error;
  }
}
```

```typescript
  field: string;
  operator: 'EQ' | 'NE' | 'IN' | 'NIN' | 'LE' | 'GE' | 'LIKE';

}

export interface ParcelSummary {
  parcelId: string;
  customerOrderId: string;
  deliveryStoreId: string;

  isPaid: boolean;
  isExpired: boolean;
  expirationDate: string;
  customer: {
    firstName: string;
    lastName: string;
    email: string | null;
  };
}
```

```typescript
// src/client/store-api-client.ts

import {
  SearchRequest,
  SearchResult,
  ParcelResponse,
  CreateParcelRequest,
  UpdateParcelRequest
} from '../types/parcel.js';

export class StoreApiClient {
  constructor(private baseUrl: string) {}

  async searchParcels(request: SearchRequest): Promise<SearchResult> {
    const response = await fetch(`${this.baseUrl}/store_parcels/search`, {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify(request),
    });

    if (!response.ok) {
      throw await this.handleError(response);
    }

    return response.json();
  }

  async getParcel(parcelId: string): Promise<ParcelResponse> {
    const response = await fetch(`${this.baseUrl}/store_parcels/parcel/${parcelId}`);

    if (!response.ok) {
      throw await this.handleError(response);
    }

    return response.json();
  }

  async createParcel(request: CreateParcelRequest): Promise<ParcelResponse> {
    const response = await fetch(`${this.baseUrl}/store_parcels/create`, {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify(request),
    });

    if (!response.ok) {
      throw await this.handleError(response);
    }

    return response.json();
  }

  async updateParcel(parcelId: string, request: UpdateParcelRequest): Promise<ParcelResponse> {
    const response = await fetch(`${this.baseUrl}/store_parcels/parcel/${parcelId}`, {
      method: 'PUT',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify(request),
    });

    if (!response.ok) {
      throw await this.handleError(response);
    }

    return response.json();
  }

  async deleteParcel(parcelId: string): Promise<void> {
    const response = await fetch(`${this.baseUrl}/store_parcels/parcel/${parcelId}`, {
      method: 'DELETE',
    });

    if (!response.ok) {
      throw await this.handleError(response);
    }
  }

  async healthCheck(): Promise<{ status: string; timestamp: number }> {
    const response = await fetch(`${this.baseUrl}/store_parcels/health`);
    return response.json();
  }

  private async handleError(response: Response): Promise<Error> {
    const body = await response.json().catch(() => ({}));
    const error = new Error(body.detail || `HTTP ${response.status}`);
    (error as any).status = response.status;
    (error as any).body = body;
    return error;
  }
}
```

## 7. Serveur MCP Principal

### 7.1 Configuration et Setup

```typescript
// src/server.ts
import { Server } from '@modelcontextprotocol/sdk/server/index.js';
import { StdioServerTransport } from '@modelcontextprotocol/sdk/server/stdio.js';
import {
  CallToolRequestSchema,
  ListToolsRequestSchema,
  ListResourcesRequestSchema,
  ReadResourceRequestSchema,
} from '@modelcontextprotocol/sdk/types.js';
import { StoreApiClient } from './client/store-api-client.js';
import {
  searchParcelsTool, handleSearchParcels,
  getParcelTool, handleGetParcel,
  createParcelTool, handleCreateParcel,
  updateParcelStatusTool, handleUpdateParcelStatus,
  deleteParcelTool, handleDeleteParcel
} from './tools/index.js';
import { storesResource, getStoresContent, statusesResource, getStatusesContent } from './resources/index.js';

const API_BASE_URL = process.env.STORE_API_URL || 'http://localhost:8080';

export async function createServer(): Promise<Server> {
  const server = new Server(
    {
      name: 'store-parcels-mcp',
      version: '1.0.0',
    },
    {
      capabilities: {
        tools: {},
        resources: {},
      },
    }
  );

  const apiClient = new StoreApiClient(API_BASE_URL);

  // List available tools
  server.setRequestHandler(ListToolsRequestSchema, async () => ({
    tools: [
      searchParcelsTool,
      getParcelTool,
      createParcelTool,
      updateParcelStatusTool,
      deleteParcelTool
    ],
  }));

  // Handle tool calls
  server.setRequestHandler(CallToolRequestSchema, async (request) => {
    const { name, arguments: args } = request.params;

    try {
      let result: string;

      switch (name) {
        case 'search_parcels':
          result = await handleSearchParcels(args as any, apiClient);
          break;
        case 'get_parcel':
          result = await handleGetParcel(args as any, apiClient);
          break;
        case 'create_parcel':
          result = await handleCreateParcel(args as any, apiClient);
          break;
        case 'update_parcel_status':
          result = await handleUpdateParcelStatus(args as any, apiClient);
          break;
        case 'delete_parcel':
          result = await handleDeleteParcel(args as any, apiClient);
          break;
        default:
          throw new Error(`Unknown tool: ${name}`);
      }

      return {
        content: [{ type: 'text', text: result }],
      };
    } catch (error: any) {
      return {
        content: [{
          type: 'text',
          text: `Error: ${error.message}`
        }],
        isError: true,
      };
    }
  });

  // List available resources
  server.setRequestHandler(ListResourcesRequestSchema, async () => ({
    resources: [storesResource, statusesResource],
  }));

  // Read resource content
  server.setRequestHandler(ReadResourceRequestSchema, async (request) => {
    const { uri } = request.params;

    let content: string;
    let mimeType: string;

    switch (uri) {
      case 'store-parcels://stores':
        content = await getStoresContent();
        mimeType = 'application/json';
        break;
      case 'store-parcels://statuses':
        content = getStatusesContent();
        mimeType = 'text/markdown';
        break;
      default:
        throw new Error(`Unknown resource: ${uri}`);
    }

    return {
      contents: [{
        uri,
        mimeType,
        text: content,
      }],
    };
  });

  return server;
}
```

